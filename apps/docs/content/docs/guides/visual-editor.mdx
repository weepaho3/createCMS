This guide shows how to compose a styled editor from `@createcms/react` and
the docs registry. For a real createCMS backend, start with the one-install
`editor-app` block in [Editor UI](/docs/guides/editor-ui): it wraps
`Editor.Root` and `Canvas.Root` and wires `useCmsDocument` /
`useCmsFieldSources`. The chrome items below (`editor-form`, `editor-canvas`,
`editor-shell`, `editor-email`) do not wrap `Root` — compose those **inside**
`Editor.Root`.

## Install

```bash
npm install @createcms/react
```

### One install (`editor-app`)

When you have a collection client, start with the CMS-wired page in
[Editor UI](/docs/guides/editor-ui):

```bash
npx shadcn@latest add https://createcms.dev/r/editor-app.json
```

### Building-block chrome

Or install the chrome items individually. Field vs Form nesting is
documented on [Editor UI](/docs/guides/editor-ui):

```bash
npx shadcn@latest add https://createcms.dev/r/editor-form.json
npx shadcn@latest add https://createcms.dev/r/editor-canvas.json
npx shadcn@latest add https://createcms.dev/r/editor-shell.json
npx shadcn@latest add https://createcms.dev/r/editor-email.json
```

The docs app hosts working demos composed from those registry items:
[live canvas](/demo/editor/pages), [form only](/demo/editor/form),
[form plus preview](/demo/editor/preview), and
[email split](/demo/editor/email).

## Editor schema and documents

The demos keep their createcms collections (`defineCollection`) and block
maps (`createBlocksMap`) and derive the editor side once, at module scope,
through the [createcms adapter](/docs/reference/react-editor-cms): the editor
compares schemas by identity.

```ts
'use client';

import {
  cmsCanvasComponents,
  fromCollection,
  toEditor,
} from '@createcms/react/editor/cms';

import { emails } from '@/app/demo/_lib/email-schema';
import { EMAIL_TREE } from '@/app/demo/_lib/email-tree';
import { pageBlocks } from '@/app/demo/_lib/pages-blocks';
import { pages } from '@/app/demo/_lib/pages-schema';
import { PAGES_TREE } from '@/app/demo/_lib/pages-tree';

// Module scope on purpose: the editor compares schemas by identity.
export const pagesSchema = fromCollection(pages);
export const pagesComponents = cmsCanvasComponents(pageBlocks);
export const PAGES_DOC = toEditor(PAGES_TREE, pages);

export const emailsSchema = fromCollection(emails);
export const EMAIL_DOC = toEditor(EMAIL_TREE, emails);
```

## Form only

The primary form demo is `CmsEditor mode="form"`: `EditorShell mode="form"`
(no palette, outline, inspector, or device toggles) plus `FormSurface` as
the single edit surface (page fields when nothing is selected). `cmsFields`
maps controls onto shadcn primitives (`Input`, `Textarea`, `Select`,
`Checkbox`). Document JSON is a `Collapsible` at the bottom of the shell,
not the primary UI.

```tsx
'use client';

import { useEditor } from '@createcms/react/editor';
import * as React from 'react';

import { createDemoCmsClient } from '@/app/demo/_lib/demo-cms-client';
import { pageBlocks } from '@/app/demo/_lib/pages-blocks';
import { PAGES_TREE } from '@/app/demo/_lib/pages-tree';
import { CmsEditor } from '@/components/editor-app';
import {
  Collapsible,
  CollapsibleContent,
  CollapsibleTrigger,
} from '@/components/ui/collapsible';

function DocumentJson() {
  const version = useEditor((state) => state.version);
  const { getDocument } = useEditor();
  const doc = getDocument();
  void version;

  return (
    <Collapsible
      data-slot="editor-document-json"
      className="border-border shrink-0 border-t"
    >
      <CollapsibleTrigger className="text-muted-foreground hover:text-foreground mt-8 flex w-full cursor-pointer items-center rounded-md border px-3 py-2 text-left text-xs font-medium">
        Document JSON
      </CollapsibleTrigger>
      <CollapsibleContent>
        <pre className="max-h-48 overflow-auto p-4 text-xs leading-relaxed">
          {JSON.stringify(doc, null, 2)}
        </pre>
      </CollapsibleContent>
    </Collapsible>
  );
}

export function FormOnlyEditor() {
  const client = React.useMemo(() => createDemoCmsClient(), []);

  return (
    <CmsEditor
      client={client}
      collection="pages"
      rootId={PAGES_TREE.blockId}
      branchId="demo"
      blocks={pageBlocks}
      mode="form"
    >
      <DocumentJson />
    </CmsEditor>
  );
}
```

## Live canvas

`Editor.Root` takes the derived `pagesSchema` and `PAGES_DOC`. `EditorShell`
children are `Canvas.Root components={pagesComponents}` with overlay rings,
toolbar, insert, drag handle, drop indicator, drag preview, and
`InlineText`. Palette and outline live in the shell's left `Sidebar`; the
inspector is the right `Sidebar`. `pagesComponents` wraps the
`createBlocksMap` components, which spread `edit` on each block root.

```tsx
'use client';

import { Editor, useEditor } from '@createcms/react/editor';
import { Canvas } from '@createcms/react/editor/canvas';
import * as React from 'react';

import {
  PAGES_DOC,
  pagesComponents,
  pagesSchema,
} from '@/app/demo/_lib/editor';
import { useDemoFieldSources } from '@/app/demo/_lib/sources';
import { useLocalDocument } from '@/app/demo/_lib/use-local-document';
import {
  BlockToolbar,
  DragPreview,
  DropIndicator,
  FieldRing,
  HoverRing,
  InlineText,
  InsertButton,
  Overlay,
  SelectionRing,
} from '@/components/editor-canvas';
import { CmsSourcesProvider, cmsFields } from '@/components/editor-form';
import { EditorShell } from '@/components/editor-shell';
import {
  Collapsible,
  CollapsibleContent,
  CollapsibleTrigger,
} from '@/components/ui/collapsible';

function DocumentJson() {
  const version = useEditor((state) => state.version);
  const { getDocument } = useEditor();
  const doc = getDocument();
  void version;

  return (
    <Collapsible
      data-slot="editor-document-json"
      className="border-border shrink-0 border-t"
    >
      <CollapsibleTrigger className="text-muted-foreground hover:text-foreground flex w-full cursor-pointer items-center px-4 py-2 text-left text-xs font-medium">
        Document JSON
      </CollapsibleTrigger>
      <CollapsibleContent>
        <pre className="max-h-48 overflow-auto p-4 text-xs leading-relaxed">
          {JSON.stringify(doc, null, 2)}
        </pre>
      </CollapsibleContent>
    </Collapsible>
  );
}

export function PagesLiveCanvas() {
  const sources = useDemoFieldSources();
  const { onChange, onSave } = useLocalDocument(PAGES_DOC);

  return (
    <div className="flex min-h-0 flex-1 flex-col overflow-hidden">
      <Editor.Root
        schema={pagesSchema}
        defaultValue={PAGES_DOC}
        onChange={onChange}
        onSave={onSave}
        fields={cmsFields}
      >
        <CmsSourcesProvider sources={sources}>
          <EditorShell>
            <Canvas.Root
              components={pagesComponents}
              style={{ position: 'relative' }}
            >
              <Overlay>
                <SelectionRing />
                <HoverRing />
                <FieldRing />
                <BlockToolbar side="top" align="start" />
                <InsertButton placement="between" type="richText" />
                <DropIndicator />
                <DragPreview />
                <InlineText />
              </Overlay>
            </Canvas.Root>
          </EditorShell>
          <DocumentJson />
        </CmsSourcesProvider>
      </Editor.Root>
    </div>
  );
}
```

<Callout type="info">
  A createCMS collection client can skip this local stand-in: install
  `editor-app` from [Editor UI](/docs/guides/editor-ui) and pass the client
  plus your `createBlocksMap` result as `blocks`. `CmsEditor` already wires `useCmsDocument` as on
  the [Editor cms adapter](/docs/reference/react-editor-cms) reference page
  (`key={doc.key}`, `onChange={doc.onChange}`, `onSave={doc.save}`, and
  `resolve={doc.resolve}` on `Canvas.Root`).
</Callout>

## Form plus preview

Root plus styled `Form` and `Editor.Preview`. Preview `render` receives the
raw editor document; `fromEditor` turns it back into the createcms tree
`BlocksRenderer` renders. For HTML or email, use `FramePreview` with
`selectable` and `render={async (doc, { signal }) => html}`. For PDF, return a `Blob` from
`FramePreview`, or render a viewer in `Preview`.

```tsx
'use client';

import { BlocksRenderer } from '@createcms/core/react/blocks';
import { Editor } from '@createcms/react/editor';
import { fromEditor } from '@createcms/react/editor/cms';

import { PAGES_DOC, pagesSchema } from '@/app/demo/_lib/editor';
import { pageBlocks } from '@/app/demo/_lib/pages-blocks';
import { pages } from '@/app/demo/_lib/pages-schema';
import { useDemoFieldSources } from '@/app/demo/_lib/sources';
import { useLocalDocument } from '@/app/demo/_lib/use-local-document';
import { CmsSourcesProvider, cmsFields, Form } from '@/components/editor-form';

export function FormPreviewEditor() {
  const sources = useDemoFieldSources();
  const { saved, onChange, onSave } = useLocalDocument(PAGES_DOC);

  return (
    <div className="flex min-h-0 flex-1 flex-col">
      <Editor.Root
        schema={pagesSchema}
        defaultValue={PAGES_DOC}
        onChange={onChange}
        onSave={onSave}
        fields={cmsFields}
      >
        <CmsSourcesProvider sources={sources}>
          <div className="grid min-h-0 flex-1 grid-cols-2 gap-4 p-4">
            <Form blockId={PAGES_DOC.root.id} />
            <Editor.Preview
              render={(doc) => (
                <BlocksRenderer
                  blocks={pageBlocks}
                  tree={fromEditor(doc, pages)}
                />
              )}
            />
          </div>
        </CmsSourcesProvider>
      </Editor.Root>
      <pre className="border-border max-h-48 overflow-auto border-t p-4 text-xs">
        {JSON.stringify(saved, null, 2)}
      </pre>
    </div>
  );
}
```

## Email split

`EditorEmail` requires a `render` prop. It owns the styled form pane and the
`Editor.FramePreview` issues panel; the consumer only supplies `render`. Do
not set `surface="frame"`.

```tsx
'use client';

import { Editor } from '@createcms/react/editor';
import { fromEditor } from '@createcms/react/editor/cms';

import { EMAIL_DOC, emailsSchema } from '@/app/demo/_lib/editor';
import { emails } from '@/app/demo/_lib/email-schema';
import { renderEmailHtml } from '@/app/demo/_lib/render-email';
import { useDemoFieldSources } from '@/app/demo/_lib/sources';
import { useLocalDocument } from '@/app/demo/_lib/use-local-document';
import { EditorEmail } from '@/components/editor-email';
import { CmsSourcesProvider, cmsFields } from '@/components/editor-form';

export function EmailSplitEditor() {
  const sources = useDemoFieldSources();
  const { saved, onChange, onSave } = useLocalDocument(EMAIL_DOC);

  return (
    <div className="flex min-h-0 flex-1 flex-col">
      <Editor.Root
        schema={emailsSchema}
        defaultValue={EMAIL_DOC}
        onChange={onChange}
        onSave={onSave}
        fields={cmsFields}
      >
        <CmsSourcesProvider sources={sources}>
          <EditorEmail
            render={async (doc) => renderEmailHtml(fromEditor(doc, emails))}
          />
        </CmsSourcesProvider>
      </Editor.Root>
      <pre className="border-border max-h-48 overflow-auto border-t p-4 text-xs">
        {JSON.stringify(saved, null, 2)}
      </pre>
    </div>
  );
}
```

## Block component with edit

Spread `edit.block` and `edit.field.<key>` on consumer block markup:

```tsx
import type { BlockComponentProps } from '@createcms/core/react/blocks';

function Hero({ properties, edit }: BlockComponentProps) {
  return (
    <section {...edit.block}>
      <h1 {...edit.field.headline}>{properties.headline}</h1>
    </section>
  );
}
```

On the published renderer, pass `edit="preview"` or `NO_EDIT` instead. See
[React](/docs/reference/react).
